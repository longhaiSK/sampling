6  Unequal Probability Sampling

6.1 Key Concepts and Formulas

Unequal Probability Sampling (UPS) improves estimation precision by sampling units with probabilities proportional to their size (PPS). If a cluster or county is larger, it has a higher chance of being included in the sample. This approach prevents extreme variance when cluster sizes vary drastically.

1. UPS With Replacement (UPSWR) When sampling with replacement, we draw \(n\) units. A single unit \(i\) has a selection probability \(\psi_i\) on each draw.

  • Total Estimator (Hansen-Hurwitz): \(\hat{t}_{pwr} = \frac{1}{n} \sum_{i=1}^n \frac{y_i}{\psi_i}\)
  • Ratio Estimator (Mean): \(\hat{\bar{y}}_{r} = \frac{\sum_{i=1}^n \frac{y_i}{\psi_i}}{\sum_{i=1}^n \frac{M_i}{\psi_i}}\)

2. UPS Without Replacement (UPSWO) When sampling without replacement, the inclusion probability for unit \(i\) across all draws is \(\pi_i\).

  • Total Estimator (Horvitz-Thompson): \(\hat{t}_{HT} = \sum_{i=1}^n \frac{y_i}{\pi_i}\)

6.2 Functions and Packages

We consolidate our custom estimation functions here, modifying them to return structured lists for cluster summaries where applicable. We also include the format_est_gt helper to render all final estimates with mathematical LaTeX headers.

# ---------------------------------------------------------
# Base Estimators
# ---------------------------------------------------------

srs_ratio_est <- function(ydata, xdata, N = Inf) { 
  n <- length(xdata)
  xbar <- mean(xdata)
  ybar <- mean(ydata)
  B_hat <- ybar / xbar
  d <- ydata - B_hat * xdata
  var_d <- sum(d^2) / (n - 1)
  sd_B_hat <- sqrt((1 - n/N) * var_d / n) / xbar
  mem <- qt(0.975, df = n - 1) * sd_B_hat
  
  output <- c(B_hat, sd_B_hat, B_hat - mem, B_hat + mem)
  names(output) <- c("Est.", "S.E.", "ci.low", "ci.upp")
  output
}

srs_reg_est <- function(ydata, xdata, xbarU, N = Inf, est.total = FALSE) {
  n <- length(ydata)
  lmfit <- lm(ydata ~ xdata)
  Bhat <- lmfit$coefficients
  efit <- lmfit$residuals
  SSe <- sum(efit^2) / (n - 2) 
  yhat_reg <- Bhat[1] + Bhat[2] * xbarU
  se_yhat_reg <- sqrt((1-n/N) * SSe / n)
  mem <- qt(0.975, df = n - 2) * se_yhat_reg
  
  output <- c(yhat_reg, se_yhat_reg, yhat_reg - mem, yhat_reg + mem)
  
  if (est.total) {
      if(!is.finite(N)) stop("N must be finite for estimating population total")
      output <- output * N
  }
  names(output) <- c("Est.", "S.E.", "ci.low", "ci.upp")
  output
}

srs_mean_est <- function(sdata, N = Inf) {
  n <- length(sdata)
  ybar <- mean(sdata)
  se.ybar <- sqrt((1 - n / N)) * sd(sdata) / sqrt(n)  
  mem <- qt(0.975, df = n - 1) * se.ybar
  output <- c(ybar, se.ybar, ybar - mem, ybar + mem)
  names(output) <- c("Est.", "S.E.", "ci.low", "ci.upp")
  output
}

# ---------------------------------------------------------
# UPS Estimators
# ---------------------------------------------------------

# Total estimation for datasets collected with UPS with Replacement
upswr_total_est <- function(total, psi) { 
  srs_mean_est(total/psi, N = Inf)
}

# Ratio estimation for datasets collected with UPS with Replacement
upswr_ratio_est <- function(total, M, psi) {
  srs_ratio_est(total/psi, M/psi, N = Inf)
}

# Cluster UPS with Replacement
cluster_upswr_ratio_est <- function(data, sid, csize, cpsi, yvar) {
    clust <- data[, sid]
    ydata <- data[, yvar]  
    
    ybari <- tapply(ydata, clust, mean)
    Mi <- tapply(data[,csize], clust, function(x) x[1])
    psi <- tapply(data[,cpsi], clust, function(x) x[1])
    t_hat_cls <- ybari * Mi 
    
    summary_df <- data.frame(
        Cluster = names(Mi),
        Mi = unname(Mi),
        psi = unname(psi),
        ybari = unname(ybari),
        ti_hat = unname(t_hat_cls),
        ti_psi = unname(t_hat_cls/psi),
        Mi_psi = unname(Mi/psi)
    )
    
    est <- srs_ratio_est(t_hat_cls/psi, Mi/psi)
    list(summary = summary_df, estimate = est)
}

# Cluster UPS without Replacement
cluster_upswo_ratio_est <- function(data, cname, csize, cpik, yvar) {
  clust <- data[, cname]
  ydata <- data[, yvar]  
  
  ybari <- tapply(ydata, clust, mean)
  Mi <- tapply(data[,csize], clust, function(x) x[1])
  pik <- tapply(data[,cpik], clust, function(x) x[1])
  t_hat_cls <- ybari * Mi 
  
  summary_df <- data.frame(
        Cluster = names(Mi),
        Mi = unname(Mi),
        pik = unname(pik),
        ybari = unname(ybari),
        ti_hat = unname(t_hat_cls)
  )
  
  est <- srs_ratio_est(t_hat_cls/pik, Mi/pik)
  list(summary = summary_df, estimate = est)
}

# ---------------------------------------------------------
# GT Formatting Helper
# ---------------------------------------------------------

format_est_gt <- function(est_data, est_type = c("mean", "total", "ratio", "reg_mean", "reg_total", "ratio_mean", "domain_mean")) {
  est_type <- match.arg(est_type)
  
  sym_map <- list(
    "mean"        = c(est = "$\\bar{y}$",         se = "$SE(\\bar{y})$"),
    "total"       = c(est = "$\\hat{t}$",         se = "$SE(\\hat{t})$"),
    "ratio"       = c(est = "$\\hat{B}$",         se = "$SE(\\hat{B})$"),
    "reg_mean"    = c(est = "$\\bar{y}_{reg}$",   se = "$SE(\\bar{y}_{reg})$"),
    "reg_total"   = c(est = "$\\hat{t}_{reg}$",   se = "$SE(\\hat{t}_{reg})$"),
    "ratio_mean"  = c(est = "$\\bar{y}_{r}$",     se = "$SE(\\bar{y}_{r})$"),
    "domain_mean" = c(est = "$\\bar{y}_{d}$",     se = "$SE(\\bar{y}_{d})$")
  )
  
  est_sym <- sym_map[[est_type]]["est"]
  se_sym  <- sym_map[[est_type]]["se"]
  
  if (is.vector(est_data) && !is.list(est_data)) {
    df <- as.data.frame(t(unname(est_data)))
  } else {
    df <- as.data.frame(est_data)
  }
  
  colnames(df) <- c("Est.", "S.E.", "ci.low", "ci.upp")
  use_stub <- !is.null(rownames(df)) && !all(rownames(df) == as.character(1:nrow(df)))
  
  df |>
    gt(rownames_to_stub = use_stub) |>
    cols_label(
      Est.   = md(est_sym),
      S.E.   = md(se_sym),
      ci.low = md("95% CI Lower"),
      ci.upp = md("95% CI Upper")
    ) |>
    fmt_number(columns = everything(), decimals = 4) |>
    tab_options(table.width = pct(70))
}

6.6 Simulation Studies for UPSWR using agpop data

Let’s simulate 2000 random samples to see how UPSWR empirically compares to Simple Random Sampling.

Code 
# Read population data
agpop <- read.csv("data/agpop.csv", na.strings = "-99")
agpop <- agpop |> filter(!is.na(acres92), !is.na(acres87))

N <- nrow(agpop)
total_acres92 <- sum(agpop$acres92)
total_acres87 <- sum(agpop$acres87)
acres87 <- agpop$acres87

# Storage vectors for simulation
est_srs300 <- rep(0, 2000)
est_srs300_reg_acres87 <- rep(0, 2000)
est_ups300_acres87 <- rep(0, 2000)

for (i in 1:2000) {
  # SRS Simulation
  srs300 <- sample(1:N, size = 300)
  acres92_srs300 <- agpop$acres92[srs300]
  acres87_srs300 <- agpop$acres87[srs300]
  
  est_srs300[i] <- (srs_mean_est(acres92_srs300, N = N) * N)[1] 
  est_srs300_reg_acres87[i] <- srs_reg_est(acres92_srs300, acres87_srs300, 
                                           total_acres87/N, N = N, est.total = TRUE)[1]
  
  # UPSWR Simulation (Probability proportional to acres87)
  ups300 <- sample(1:N, size = 300, prob = acres87, replace = TRUE)
  acres92_ups300 <- agpop$acres92[ups300]
  
  psi <- (acres87 / sum(acres87))[ups300]
  est_ups300_acres87[i] <- upswr_total_est(acres92_ups300, psi)[1] 
}
